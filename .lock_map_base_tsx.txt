    38	function severityColor(sev: number) {
    39	  if (sev === 3) return "#ef4444";
    40	  if (sev === 2) return "#f59e0b";
    41	  return "#10b981";

    43	
    44	function safeStatus(h: any): "reported" | "resolved" {
    45	  return h?.status === "resolved" ? "resolved" : "reported";
    46	}
    47	
    48	function shortTypeLabel(t: string) {
    49	  const found = HAZARD_TYPES.find((x) => x.key === t);
    50	  return found ? found.label : String(t || "Hazard");

   602	          style={{
   603	            flex: 1,
   604	            backgroundColor: "rgba(148,163,184,0.20)",
   605	            paddingVertical: 12,
   606	            borderRadius: 14,
   607	            alignItems: "center",
   608	          }}
   609	          disabled={isMutating}
   610	        >
   611	          <Text style={{ color: "white", fontWeight: "900" }}>CLOSE</Text>
   612	        </Pressable>
   613	
   614	        {safeStatus(selected) !== "resolved" ? (
   615	          <Pressable
   616	            onPress={resolveSelected}
   617	            style={{
   618	              flex: 1,
   619	              backgroundColor: "#16a34a",
   620	              paddingVertical: 12,
   621	              borderRadius: 14,
   622	              alignItems: "center",
   623	              opacity: isMutating ? 0.7 : 1,
   624	            }}
   625	            disabled={isMutating}
   626	          >
   627	            <Text style={{ color: "white", fontWeight: "900" }}>
   628	              {isMutating ? "…" : "RESOLVE"}
   629	            </Text>
   630	          </Pressable>
   631	        ) : (
   632	          <View style={{ flex: 1 }} />
   633	        )}
   634	
   635	        <Pressable
   636	          onPress={deleteSelected}
   637	          style={{
   638	            flex: 1,
   639	            backgroundColor: "#991b1b",
   640	            paddingVertical: 12,
   641	            borderRadius: 14,
   642	            alignItems: "center",
   643	            opacity: isMutating ? 0.7 : 1,
   644	          }}
   645	          disabled={isMutating}
   646	        >
   647	          <Text style={{ color: "white", fontWeight: "900" }}>
   648	            {isMutating ? "…" : "DELETE"}
   649	          </Text>
   650	        </Pressable>
   651	      </View>
   652	    </View>
   653	  ) : null;
   654	
   655	  return (
   656	    <View style={{ flex: 1 }}>
   657	      <MapView onRegionChangeComplete={setRegion}
   658	        ref={(r) => (mapRef.current = r)}
   659	        style={{ flex: 1 }}
   660	        initialRegion={region}
   661	        onLongPress={(e: MapPressEvent) => {
   662	          // LOCKED: long-press primary trigger
   663	          if (!isMutating) enterPlacingMode();
   664	        }}
   665	        showsUserLocation={true}
   666	        showsMyLocationButton={false}
   667	      >
   668	        {visibleHazards.map((h: any) => {
   669	          const st = safeStatus(h);
   670	          const pinColor = st === "resolved" ? "#64748b" : severityColor(h.severity);
   671	          const opacity = st === "resolved" ? 0.25 : 1.0;
   672	
   673	          // Remount marker when status changes to avoid reuse bleed
   674	          const markerKey = `${h.id}-${st}`;
   675	
   676	          return (
   677	            <Marker
   678	              key={markerKey}
   679	              coordinate={{ latitude: h.latitude, longitude: h.longitude }}
   680	              pinColor={pinColor}
   681	              opacity={opacity}
   682	              tracksViewChanges={false}
   683	              onPress={() => onMarkerPress(String(h.id))}
   684	            />
   685	          );
   686	        })}
   687	      </MapView>
   688	
   689	      {topBar}
   690	      {recenterBtn}
   691	      {placementMode ? placingOverlay : selected ? selectedCard : bottomReport}
   692	    </View>
   693	  );
