"use client";

type Hazard = {
  id: string;
  type: string | null;
  severity: "HIGH" | "MEDIUM" | "LOW";
  status: "reported" | "resolved";
  created_at: string;
  city?: string | null;
  latitude?: number | null;
  longitude?: number | null;
};

function daysOpen(createdAt: string) {
  const created = new Date(createdAt);
  const now = new Date();
  return Math.floor((now.getTime() - created.getTime()) / (1000 * 60 * 60 * 24));
}

function displayStatus(status: Hazard["status"]) {
  return status === "reported" ? "ACTIVE" : "RESOLVED";
}

function rowStyle(h: Hazard): React.CSSProperties {
  const base: React.CSSProperties = {
    borderBottom: "1px solid #e5e7eb",
  };

  if (h.status === "resolved") {
    return {
      ...base,
      opacity: 0.45,
      filter: "grayscale(60%)",
    };
  }

  if (h.severity === "HIGH") {
    return {
      ...base,
      background: "#dc262640",
      borderLeft: "6px solid #dc2626",
    };
  }

  if (h.severity === "MEDIUM") {
    return {
      ...base,
      background: "#f59e0b33",
      borderLeft: "6px solid #f59e0b",
    };
  }

  return {
    ...base,
    background: "#16a34a33",
    borderLeft: "6px solid #16a34a",
  };
}

export default function HazardReportTable({ hazards }: { hazards: Hazard[] }) {
  return (
    <div style={{ marginTop: 32 }}>
      <h2 style={{ fontWeight: 800, fontSize: 18, marginBottom: 6 }}>
        Hazard Reports
      </h2>

      <div style={{ fontSize: 13, opacity: 0.85, marginBottom: 10 }}>
        <span style={{ color: "#dc2626", fontWeight: 700 }}>● HIGH</span> – urgent / critical &nbsp;
        <span style={{ color: "#f59e0b", fontWeight: 700 }}>● MEDIUM</span> – needs attention &nbsp;
        <span style={{ color: "#16a34a", fontWeight: 700 }}>● LOW</span> – monitor
      </div>

      <div style={{ overflowX: "auto" }}>
        <table style={{ width: "100%", borderCollapse: "collapse", fontSize: 14 }}>
          <thead>
            <tr style={{ background: "#111", color: "white" }}>
              <th style={th}>Severity</th>
              <th style={th}>Type</th>
              <th style={th}>Status</th>
              <th style={th}>Reported</th>
              <th style={th}>Days Open</th>
              <th style={th}>City</th>
              <th style={th}>Latitude</th>
              <th style={th}>Longitude</th>
            </tr>
          </thead>
          <tbody>
            {hazards.map(h => {
              const openDays =
                h.status === "resolved" ? null : daysOpen(h.created_at);

              return (
                <tr key={h.id} style={rowStyle(h)}>
                  <td style={td}>{h.severity}</td>
                  <td style={td}>{h.type ?? "-"}</td>
                  <td style={td}>{displayStatus(h.status)}</td>
                  <td style={td}>
                    {new Date(h.created_at).toLocaleString()}
                  </td>
                  <td
                    style={{
                      ...td,
                      fontWeight: openDays && openDays > 0 ? 800 : 400,
                      color: openDays && openDays > 0 ? "#dc2626" : undefined,
                    }}
                  >
                    {openDays === null ? "-" : openDays}
                  </td>
                  <td style={td}>{h.city ?? "-"}</td>
                  <td style={td}>{h.latitude ?? "-"}</td>
                  <td style={td}>{h.longitude ?? "-"}</td>
                </tr>
              );
            })}
          </tbody>
        </table>
      </div>
    </div>
  );
}

const th: React.CSSProperties = {
  textAlign: "left",
  padding: 10,
  fontWeight: 700,
};

const td: React.CSSProperties = {
  padding: 10,
  whiteSpace: "nowrap",
};
