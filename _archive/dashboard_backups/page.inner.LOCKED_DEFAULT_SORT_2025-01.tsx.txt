"use client";

import { useEffect, useState } from "react";
import Link from "next/link";
import { supabase } from "@/lib/supabase";
import HazardReportTable from "./components/HazardReportTable";

type Severity = "HIGH" | "MEDIUM" | "LOW";
type Status = "reported" | "resolved";

type Hazard = {
  id: string;
  type: string | null;
  severity: Severity;
  status: Status;
  created_at: string;
  city?: string | null;
  latitude?: number | null;
  longitude?: number | null;
};

const SEVERITY_RANK: Record<Severity, number> = {
  HIGH: 0,
  MEDIUM: 1,
  LOW: 2,
};

export default function DashboardInner() {
  const [hazards, setHazards] = useState<Hazard[]>([]);
  const [loading, setLoading] = useState(true);

  const [severityFilter, setSeverityFilter] =
    useState<Severity | "ALL">("ALL");
  const [statusFilter, setStatusFilter] =
    useState<"ALL" | "ACTIVE" | "RESOLVED">("ALL");

  useEffect(() => {
    supabase
      .from("hazards")
      .select("*")
      .then(({ data }) => {
        setHazards((data || []) as Hazard[]);
        setLoading(false);
      });
  }, []);

  const filteredHazards = hazards
    .filter(h => {
      const severityOk =
        severityFilter === "ALL" || h.severity === severityFilter;

      const statusOk =
        statusFilter === "ALL" ||
        (statusFilter === "ACTIVE" && h.status === "reported") ||
        (statusFilter === "RESOLVED" && h.status === "resolved");

      return severityOk && statusOk;
    })
    .sort((a, b) => {
      // ACTIVE first
      if (a.status !== b.status) {
        return a.status === "reported" ? -1 : 1;
      }

      // Severity priority
      if (a.severity !== b.severity) {
        return SEVERITY_RANK[a.severity] - SEVERITY_RANK[b.severity];
      }

      // Newest first
      return (
        new Date(b.created_at).getTime() -
        new Date(a.created_at).getTime()
      );
    });

  const count = (sev: Severity) =>
    hazards.filter(h => h.severity === sev && h.status === "reported").length;

  const active = hazards.filter(h => h.status === "reported").length;
  const resolved = hazards.filter(h => h.status === "resolved").length;

  return (
    <main style={{ padding: 24 }}>
      <h1 style={{ fontSize: 28, fontWeight: 900, marginBottom: 12 }}>
        LGU Hazard Ops
      </h1>

      {/* SUMMARY */}
      <div style={{ display: "flex", gap: 16, marginBottom: 16 }}>
        <Stat label="HIGH" value={count("HIGH")} color="#dc2626" />
        <Stat label="MED" value={count("MEDIUM")} color="#f59e0b" />
        <Stat label="LOW" value={count("LOW")} color="#16a34a" />
      </div>

      <div style={{ marginBottom: 20, opacity: 0.8 }}>
        Active: {active} • Resolved: {resolved}
      </div>

      {/* ACTIONS */}
      <div style={{ display: "flex", gap: 12, marginBottom: 16 }}>
        <Link
          href="/dashboard/map"
          style={{
            background: "#2563eb",
            color: "#fff",
            padding: "10px 16px",
            borderRadius: 8,
            fontWeight: 800,
            textDecoration: "none",
          }}
        >
          View Map →
        </Link>
      </div>

      {/* FILTERS */}
      <div
        style={{
          display: "flex",
          gap: 12,
          alignItems: "center",
          marginBottom: 16,
          flexWrap: "wrap",
        }}
      >
        <FilterGroup label="Severity">
          {["ALL", "HIGH", "MEDIUM", "LOW"].map(v => (
            <FilterButton
              key={v}
              active={severityFilter === v}
              onClick={() => setSeverityFilter(v as any)}
            >
              {v}
            </FilterButton>
          ))}
        </FilterGroup>

        <FilterGroup label="Status">
          {["ALL", "ACTIVE", "RESOLVED"].map(v => (
            <FilterButton
              key={v}
              active={statusFilter === v}
              onClick={() => setStatusFilter(v as any)}
            >
              {v}
            </FilterButton>
          ))}
        </FilterGroup>

        <button
          onClick={() => {
            setSeverityFilter("ALL");
            setStatusFilter("ALL");
          }}
          style={{
            marginLeft: "auto",
            background: "#111",
            color: "#fff",
            padding: "8px 12px",
            borderRadius: 6,
            fontWeight: 700,
            border: "none",
            cursor: "pointer",
          }}
        >
          Reset
        </button>
      </div>

      {/* TABLE */}
      {loading ? (
        <div>Loading reports…</div>
      ) : (
        <HazardReportTable hazards={filteredHazards} />
      )}
    </main>
  );
}

function FilterGroup({
  label,
  children,
}: {
  label: string;
  children: React.ReactNode;
}) {
  return (
    <div style={{ display: "flex", gap: 6, alignItems: "center" }}>
      <span style={{ fontWeight: 700, fontSize: 13 }}>{label}:</span>
      {children}
    </div>
  );
}

function FilterButton({
  active,
  onClick,
  children,
}: {
  active: boolean;
  onClick: () => void;
  children: React.ReactNode;
}) {
  return (
    <button
      onClick={onClick}
      style={{
        background: active ? "#2563eb" : "#1f2937",
        color: "#fff",
        padding: "6px 10px",
        borderRadius: 6,
        border: "none",
        fontWeight: 700,
        cursor: "pointer",
        fontSize: 12,
      }}
    >
      {children}
    </button>
  );
}

function Stat({
  label,
  value,
  color,
}: {
  label: string;
  value: number;
  color: string;
}) {
  return (
    <div
      style={{
        background: color + "22",
        border: `2px solid ${color}`,
        padding: 16,
        borderRadius: 12,
        minWidth: 120,
      }}
    >
      <div style={{ fontWeight: 900 }}>{label}</div>
      <div style={{ fontSize: 28, fontWeight: 900 }}>{value}</div>
    </div>
  );
}
