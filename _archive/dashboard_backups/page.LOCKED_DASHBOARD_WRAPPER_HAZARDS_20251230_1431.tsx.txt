import { Suspense } from "react";
import DashboardInner from "./page.inner";
import { supabase } from "@/lib/supabase";

export default async function DashboardPage() {
  const res = await supabase
    .from("hazards")
    .select("id,type,severity,status,latitude,longitude,created_at");

  const hazards = res.data ?? [];

  return (
    <Suspense fallback={<div style={{ padding: 24 }}>Loading dashboard…</div>}>
      <DashboardInner hazards={hazards} />
    </Suspense>
  );
}
