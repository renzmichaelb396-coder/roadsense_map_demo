type Severity = "HIGH" | "MEDIUM" | "LOW";

type Hazard = {
  id: string;
  type: string;
  severity: Severity;
  status: "reported" | "resolved";
  created_at?: string;
};

function displayStatus(status: Hazard["status"]) {
  return status === "reported" ? "ACTIVE" : "RESOLVED";
}

export default function HazardReportTable({
  hazards,
  getActionLabel,
}: {
  hazards: Hazard[];
  getActionLabel?: (h: Hazard) =>
    | "Needs Immediate Action"
    | "Schedule Repair"
    | "Monitor"
    | "Resolved";
}) {
  return (
    <table
      style={{
        width: "100%",
        borderCollapse: "collapse",
        fontSize: 14,
      }}
    >
      <thead>
        <tr>
          <th align="left">Type</th>
          <th align="left">Severity</th>
          <th align="left">Status</th>
          {getActionLabel && <th align="left">Action Required</th>}
        </tr>
      </thead>
      <tbody>
        {hazards.map((h) => {
          const severityStyle =
            h.severity === "HIGH"
              ? {
                  borderLeft: "6px solid #dc2626",
                  fontWeight: 700,
                  background: "#fee2e2",
                }
              : h.severity === "MEDIUM"
              ? {
                  borderLeft: "4px solid #f59e0b",
                  background: "#fef3c7",
                }
              : {
                  borderLeft: "2px solid #16a34a",
                  background: "#ecfdf5",
                  opacity: 0.85,
                };

          return (
            <tr key={h.id} style={severityStyle}>
              <td>{h.type}</td>
              <td>{h.severity}</td>
              <td>{displayStatus(h.status)}</td>
              {getActionLabel && <td>{getActionLabel(h)}</td>}
            </tr>
          );
        })}
      </tbody>
    </table>
  );
}
