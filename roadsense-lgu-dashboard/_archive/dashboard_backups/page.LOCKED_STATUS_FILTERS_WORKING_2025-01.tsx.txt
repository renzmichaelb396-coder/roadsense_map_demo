"use client";

import "mapbox-gl/dist/mapbox-gl.css";

import { useEffect, useRef, useState } from "react";
import mapboxgl from "mapbox-gl";
import { supabase } from "@/lib/supabase";

mapboxgl.accessToken = process.env.NEXT_PUBLIC_MAPBOX_TOKEN!;

type Severity = "HIGH" | "MEDIUM" | "LOW";
type RawStatus = "reported" | "resolved" | "ACTIVE" | "RESOLVED";
type StatusFilter = "ALL" | "ACTIVE" | "RESOLVED";

type Hazard = {
  type: string | null;
  severity: Severity;
  status: RawStatus;
  latitude: number;
  longitude: number;
  created_at: string;
};

const SEVERITY_COLOR: Record<Severity, string> = {
  HIGH: "#dc2626",
  MEDIUM: "#f59e0b",
  LOW: "#16a34a",
};

function normalizeStatus(status: RawStatus): "ACTIVE" | "RESOLVED" {
  if (status === "reported" || status === "ACTIVE") return "ACTIVE";
  return "RESOLVED";
}

function hazardLetter(type: string | null) {
  const t = (type || "").toLowerCase();
  if (t === "pothole") return "P";
  if (t === "crack") return "C";
  if (t === "construction") return "R";
  if (t === "flood") return "F";
  if (t === "debris") return "D";
  return "?";
}

function createLGUMarker(h: Hazard) {
  const el = document.createElement("div");

  el.style.cssText = `
    width:40px;
    height:40px;
    border-radius:999px;
    background:${SEVERITY_COLOR[h.severity]};
    color:#ffffff;
    font-weight:900;
    font-size:18px;
    display:flex;
    align-items:center;
    justify-content:center;
    border:3px solid #ffffff;
    box-shadow:
      0 0 0 2px rgba(0,0,0,0.55),
      0 6px 14px rgba(0,0,0,0.45);
    opacity:${normalizeStatus(h.status) === "RESOLVED" ? 0.45 : 1};
    cursor:pointer;
    user-select:none;
  `;

  el.textContent = hazardLetter(h.type);
  return el;
}

export default function DashboardMap() {
  const containerRef = useRef<HTMLDivElement | null>(null);
  const mapRef = useRef<mapboxgl.Map | null>(null);
  const markersRef = useRef<mapboxgl.Marker[]>([]);

  const [hazards, setHazards] = useState<Hazard[]>([]);
  const [statusFilter, setStatusFilter] = useState<StatusFilter>("ALL");

  // INIT MAP
  useEffect(() => {
    const map = new mapboxgl.Map({
      container: containerRef.current!,
      style: "mapbox://styles/mapbox/light-v11",
      center: [120.9842, 14.5995],
      zoom: 12,
      minZoom: 10,
    });

    map.dragRotate.disable();
    map.touchZoomRotate.disableRotation();
    mapRef.current = map;

    supabase
      .from("hazards")
      .select("type,severity,status,latitude,longitude,created_at")
      .then(({ data }) => {
        if (data) setHazards(data as Hazard[]);
      });

    return () => map.remove();
  }, []);

  // REBUILD MARKERS
  useEffect(() => {
    if (!mapRef.current) return;

    markersRef.current.forEach(m => m.remove());
    markersRef.current = [];

    hazards.forEach(h => {
      if (!h.latitude || !h.longitude) return;

      const uiStatus = normalizeStatus(h.status);

      if (statusFilter !== "ALL" && uiStatus !== statusFilter) return;

      const marker = new mapboxgl.Marker({
        element: createLGUMarker(h),
      })
        .setLngLat([h.longitude, h.latitude])
        .addTo(mapRef.current!);

      markersRef.current.push(marker);
    });
  }, [hazards, statusFilter]);

  return (
    <>
      <div
        style={{
          position: "absolute",
          zIndex: 10,
          top: 16,
          left: 16,
          display: "flex",
          gap: 8,
          background: "#0b0b0b",
          padding: 8,
          borderRadius: 10,
        }}
      >
        {["ALL", "ACTIVE", "RESOLVED"].map(v => (
          <button
            key={v}
            onClick={() => setStatusFilter(v as StatusFilter)}
            style={{
              background: statusFilter === v ? "#2563eb" : "#1f2937",
              color: "#fff",
              border: "none",
              borderRadius: 6,
              padding: "6px 10px",
              fontWeight: 800,
              cursor: "pointer",
              fontSize: 12,
            }}
          >
            {v}
          </button>
        ))}
      </div>

      <div ref={containerRef} style={{ width: "100%", height: "100vh" }} />
    </>
  );
}
