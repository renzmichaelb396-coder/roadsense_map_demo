"use client";

import "mapbox-gl/dist/mapbox-gl.css";

import { useEffect, useRef, useState } from "react";
import mapboxgl from "mapbox-gl";
import { supabase } from "@/lib/supabase";

mapboxgl.accessToken = process.env.NEXT_PUBLIC_MAPBOX_TOKEN!;

type Severity = "HIGH" | "MEDIUM" | "LOW";
type Status = "reported" | "resolved" | "active";
type StatusFilter = "ALL" | "ACTIVE" | "RESOLVED";
type SeverityFilter = "ALL" | Severity;

type Hazard = {
  type: string | null;
  severity: Severity;
  status: string;
  latitude: number;
  longitude: number;
  created_at: string;
};

const SEVERITY_COLOR: Record<Severity, string> = {
  HIGH: "#dc2626",
  MEDIUM: "#f59e0b",
  LOW: "#16a34a",
};

function normalizeStatus(s: string) {
  return s.toLowerCase();
}

function hazardLetter(type: string | null) {
  const t = (type || "").toLowerCase();
  if (t === "pothole") return "P";
  if (t === "crack") return "C";
  if (t === "construction") return "R";
  if (t === "flood") return "F";
  if (t === "debris") return "D";
  return "?";
}

function createLGUMarker(h: Hazard) {
  const el = document.createElement("div");

  el.style.cssText = `
    width:40px;
    height:40px;
    border-radius:999px;
    background:${SEVERITY_COLOR[h.severity]};
    color:#ffffff;
    font-weight:900;
    font-size:18px;
    display:flex;
    align-items:center;
    justify-content:center;
    border:3px solid #ffffff;
    box-shadow:
      0 0 0 2px rgba(0,0,0,0.55),
      0 6px 14px rgba(0,0,0,0.45);
    opacity:${normalizeStatus(h.status) === "resolved" ? 0.45 : 1};
    cursor:pointer;
    user-select:none;
  `;

  el.textContent = hazardLetter(h.type);
  return el;
}

function downloadCSV(rows: Hazard[]) {
  if (!rows.length) return;

  const headers = [
    "type",
    "severity",
    "status",
    "latitude",
    "longitude",
    "created_at",
  ];

  const csv = [
    headers.join(","),
    ...rows.map(r =>
      headers.map(h => JSON.stringify((r as any)[h] ?? "")).join(",")
    ),
  ].join("\n");

  const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
  const url = URL.createObjectURL(blob);

  const a = document.createElement("a");
  a.href = url;
  a.download = `hazards_${Date.now()}.csv`;
  a.click();

  URL.revokeObjectURL(url);
}

export default function DashboardMap() {
  const containerRef = useRef<HTMLDivElement | null>(null);
  const mapRef = useRef<mapboxgl.Map | null>(null);
  const markersRef = useRef<mapboxgl.Marker[]>([]);

  const [hazards, setHazards] = useState<Hazard[]>([]);
  const [statusFilter, setStatusFilter] = useState<StatusFilter>("ALL");
  const [severityFilter, setSeverityFilter] = useState<SeverityFilter>("ALL");

  useEffect(() => {
    const map = new mapboxgl.Map({
      container: containerRef.current!,
      style: "mapbox://styles/mapbox/light-v11",
      center: [120.9842, 14.5995],
      zoom: 12,
      minZoom: 10,
    });

    map.dragRotate.disable();
    map.touchZoomRotate.disableRotation();

    mapRef.current = map;

    supabase
      .from("hazards")
      .select("type,severity,status,latitude,longitude,created_at")
      .then(({ data }) => {
        if (data) setHazards(data as Hazard[]);
      });

    return () => map.remove();
  }, []);

  const filteredHazards = hazards.filter(h => {
    const status = normalizeStatus(h.status);

    if (statusFilter === "ACTIVE" && status !== "reported" && status !== "active")
      return false;
    if (statusFilter === "RESOLVED" && status !== "resolved") return false;
    if (severityFilter !== "ALL" && h.severity !== severityFilter) return false;

    return true;
  });

  useEffect(() => {
    if (!mapRef.current) return;

    markersRef.current.forEach(m => m.remove());
    markersRef.current = [];

    filteredHazards.forEach(h => {
      if (!h.latitude || !h.longitude) return;

      const marker = new mapboxgl.Marker({
        element: createLGUMarker(h),
      })
        .setLngLat([h.longitude, h.latitude])
        .addTo(mapRef.current!);

      markersRef.current.push(marker);
    });
  }, [filteredHazards]);

  const statusCounts = {
    ALL: hazards.length,
    ACTIVE: hazards.filter(h => {
      const s = normalizeStatus(h.status);
      return s === "reported" || s === "active";
    }).length,
    RESOLVED: hazards.filter(h => normalizeStatus(h.status) === "resolved").length,
  };

  const severityCounts = {
    ALL: hazards.length,
    HIGH: hazards.filter(h => h.severity === "HIGH").length,
    MEDIUM: hazards.filter(h => h.severity === "MEDIUM").length,
    LOW: hazards.filter(h => h.severity === "LOW").length,
  };

  return (
    <>
      <div
        style={{
          position: "absolute",
          zIndex: 10,
          top: 16,
          left: 16,
          display: "flex",
          gap: 8,
          background: "#0b0b0b",
          padding: 8,
          borderRadius: 10,
          alignItems: "center",
        }}
      >
        {(["ALL", "ACTIVE", "RESOLVED"] as StatusFilter[]).map(v => (
          <button
            key={v}
            onClick={() => setStatusFilter(v)}
            style={{
              background: statusFilter === v ? "#2563eb" : "#1f2937",
              color: "#fff",
              border: "none",
              borderRadius: 6,
              padding: "6px 10px",
              fontWeight: 800,
              fontSize: 12,
              cursor: "pointer",
            }}
          >
            {v} ({statusCounts[v]})
          </button>
        ))}

        {(["ALL", "HIGH", "MEDIUM", "LOW"] as SeverityFilter[]).map(v => (
          <button
            key={v}
            onClick={() => setSeverityFilter(v)}
            style={{
              background: severityFilter === v ? "#2563eb" : "#1f2937",
              color: "#fff",
              border: "none",
              borderRadius: 6,
              padding: "6px 10px",
              fontWeight: 800,
              fontSize: 12,
              cursor: "pointer",
            }}
          >
            {v} ({severityCounts[v]})
          </button>
        ))}

        <button
          onClick={() => downloadCSV(filteredHazards)}
          style={{
            marginLeft: 12,
            background: "#111827",
            color: "#fff",
            border: "1px solid #374151",
            borderRadius: 6,
            padding: "6px 12px",
            fontWeight: 800,
            fontSize: 12,
            cursor: "pointer",
          }}
        >
          Download CSV
        </button>
      </div>

      <div ref={containerRef} style={{ width: "100%", height: "100vh" }} />
    </>
  );
}
