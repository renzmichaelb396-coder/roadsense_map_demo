"use client";

import "mapbox-gl/dist/mapbox-gl.css";

import { useEffect, useRef, useState } from "react";
import mapboxgl from "mapbox-gl";
import { supabase } from "@/lib/supabase";

mapboxgl.accessToken = process.env.NEXT_PUBLIC_MAPBOX_TOKEN!;

type Severity = "HIGH" | "MEDIUM" | "LOW";
type Status = "reported" | "resolved";

type Hazard = {
  type: string | null;
  severity: Severity;
  status: Status;
  latitude: number;
  longitude: number;
  created_at: string;
};

const SEVERITY_COLOR: Record<Severity, string> = {
  HIGH: "#dc2626",
  MEDIUM: "#f59e0b",
  LOW: "#16a34a",
};

function hazardLetter(type: string | null) {
  const t = (type || "").toLowerCase();
  if (t === "pothole") return "P";
  if (t === "crack") return "C";
  if (t === "construction") return "R";
  return "?";
}

function createLGUMarker(h: {
  type: string | null;
  severity: Severity;
  status: Status;
}) {
  const el = document.createElement("div");

  el.style.cssText = `
    width:40px;
    height:40px;
    border-radius:999px;
    background:${SEVERITY_COLOR[h.severity]};
    color:#fff;
    font-weight:900;
    font-size:18px;
    display:flex;
    align-items:center;
    justify-content:center;
    border:3px solid #fff;
    box-shadow:0 6px 14px rgba(0,0,0,0.45);
    opacity:${h.status === "resolved" ? 0.45 : 1};
    cursor:pointer;
  `;

  el.textContent = hazardLetter(h.type);
  return el;
}

export default function DashboardMap() {
  const mapRef = useRef<HTMLDivElement | null>(null);
  const [hazards, setHazards] = useState<Hazard[]>([]);

  useEffect(() => {
    const params = new URLSearchParams(window.location.search);
    const lat = Number(params.get("lat"));
    const lng = Number(params.get("lng"));

    const center =
      !isNaN(lat) && !isNaN(lng) ? [lng, lat] : [120.9842, 14.5995];

    const zoom = !isNaN(lat) && !isNaN(lng) ? 15 : 11;

    const map = new mapboxgl.Map({
      container: mapRef.current!,
      style: "mapbox://styles/mapbox/streets-v11",
      center,
      zoom,
    });

    supabase
      .from("hazards")
      .select("type,severity,status,latitude,longitude,created_at")
      .then(({ data }) => {
        if (!data) return;

        setHazards(data as Hazard[]);

        data.forEach(h => {
          if (!h.latitude || !h.longitude) return;

          new mapboxgl.Marker({
            element: createLGUMarker(h),
          })
            .setLngLat([h.longitude, h.latitude])
            .addTo(map);
        });
      });

    return () => map.remove();
  }, []);

  return <div ref={mapRef} style={{ width: "100%", height: "100vh" }} />;
}
